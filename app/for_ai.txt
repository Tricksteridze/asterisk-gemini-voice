import socket
import threading
import speech_recognition as sr
import os
import time
from google import genai
from gtts import gTTS

api_key = os.environ.get("GEMINI_API_KEY")
ai_client = genai.Client(api_key=api_key) if api_key else None

LANG_MAP = {
    "1": {"stt": "en-US", "tts": "en", "err": "System error, please try again later."},
    "2": {"stt": "ru-RU", "tts": "ru", "err": "Ошибка системы, попробуйте позже."}
}


def prepare_menu_audio():
    if not os.path.exists("/tmp/intro_en.mp3"):
        gTTS(text="For English, press one.", lang="en").save("/tmp/intro_en.mp3")
    if not os.path.exists("/tmp/intro_ru.mp3"):
        gTTS(text="Для выбора русского языка нажмите два.", lang="ru").save("/tmp/intro_ru.mp3")
    if not os.path.exists("/tmp/menu.wav"):
        os.system(
            "ffmpeg -y -i 'concat:/tmp/intro_en.mp3|/tmp/intro_ru.mp3' -ar 8000 -ac 1 -c:a pcm_s16le /tmp/menu.wav > /dev/null 2>&1")


def transcribe_audio(file_path, lang_code):
    print(f"[STT] Processing file: {file_path} with lang {lang_code}")
    r = sr.Recognizer()

    if not os.path.exists(file_path):
        print(f"[STT] Error: File {file_path} not found")
        return None

    with sr.AudioFile(file_path) as source:
        audio_data = r.record(source)
        try:
            text = r.recognize_google(audio_data, language=lang_code)
            print(f"[STT] Recognized text: {text}")
            return text
        except sr.UnknownValueError:
            print("[STT] Speech unrecognized or silence")
        except sr.RequestError as e:
            print(f"[STT] Error from service: {e}")
    return None


def generate_tts(text, lang, output_path):
    print(f"[TTS] Generating audio for text: {text} with lang: {lang}")
    try:
        tts = gTTS(text=text, lang=lang, slow=False)
        temp_mp3 = output_path + ".mp3"
        tts.save(temp_mp3)
        os.system(f"ffmpeg -y -i {temp_mp3} -ar 8000 -ac 1 -c:a pcm_s16le {output_path}.wav > /dev/null 2>&1")
        if os.path.exists(temp_mp3):
            os.remove(temp_mp3)
        print(f"[TTS] Audio saved to {output_path}.wav")
        return True
    except Exception as e:
        print(f"[TTS] Error during generation: {e}")
        return False


def get_dtmf_choice(client_socket, f):
    client_socket.sendall(b"GET DATA /tmp/menu 5000 1\n")
    response = f.readline().strip()
    print(f"[Asterisk] Menu response: {response}")

    if "result=" in response:
        digit = response.split("result=")[1].split()[0]
        if digit in ["49", "1"]:
            return "1"
        if digit in ["50", "2"]:
            return "2"
    return None


def handle_asterisk_call(client_socket):
    print("\n[Python] Call connected")
    f = client_socket.makefile(mode='r', encoding='utf-8')

    while True:
        line = f.readline()
        if not line:
            break
        if line.strip() == "":
            break

    client_socket.sendall(b"ANSWER\n")
    f.readline()

    prepare_menu_audio()

    selected_digit = get_dtmf_choice(client_socket, f)
    if not selected_digit:
        print("[Python] No valid DTMF received. Defaulting to Russian (2)")
        selected_digit = "2"

    config = LANG_MAP[selected_digit]
    print(f"[Python] Target language established: {config['stt']}")

    chat = None
    if ai_client:
        sys_instr = (
            "Ты — продвинутый ИИ-ассистент, полностью осознающий свою техническую природу. "
            "Ты запущен внутри Docker-контейнера в Python-скрипте FastAGI, который управляется сервером телефонии Asterisk. "
            "Ты разговариваешь с пользователем в режиме реального времени по телефонному звонку. "
            "Отвечай кратко, живо, емко — максимум 2 предложения. Никогда не используй markdown-разметку (звездочки, решетки). "
            "Отвечай строго на том языке, на котором к тебе обратились."
        )
        chat = ai_client.chats.create(
            model='gemini-2.5-flash',
            config={"system_instruction": sys_instr}
        )

    while True:
        if os.path.exists("/tmp/user_voice.wav"):
            os.remove("/tmp/user_voice.wav")

        client_socket.sendall(b"EXEC Playback beep\n")
        f.readline()

        print("[Python] Listening for user speech...")
        client_socket.sendall(b"EXEC Record /tmp/user_voice.wav,1.5,120,y\n")
        record_response = f.readline().strip()
        print(f"[Asterisk] Record response: {record_response}")

        if not record_response or "result=-1" in record_response or "hangup" in record_response.lower():
            print("[Python] User hung up the phone.")
            break

        user_text = transcribe_audio("/tmp/user_voice.wav", config["stt"])

        if not user_text:
            print("[Python] Silence detected or could not understand. Ending call loop.")
            break

        tts_text = None
        if chat:
            for attempt in range(3):
                try:
                    print(f"[LLM] User (Attempt {attempt+1}): {user_text}")
                    response = chat.send_message(user_text)
                    tts_text = response.text.strip()
                    print(f"[LLM] Gemini: {tts_text}")
                    break
                except Exception as e:
                    print(f"[LLM] Attempt {attempt+1} failed with error: {e}")
                    if attempt < 2:
                        print("[LLM] Waiting 1.5 seconds before next retry...")
                        time.sleep(1.5)

        if not tts_text:
            tts_text = config["err"]

        if generate_tts(tts_text, config["tts"], "/tmp/ai_response"):
            print("[Python] Streaming response back to Asterisk...")
            client_socket.sendall(b"STREAM FILE /tmp/ai_response #\n")
            stream_resp = f.readline().strip()
            if not stream_resp or "result=-1" in stream_resp:
                print("[Python] User hung up while AI was speaking.")
                break

    print("[Python] Hanging up the call and tearing down")
    client_socket.sendall(b"HANGUP\n")
    try:
        f.readline()
    except Exception:
        pass

    f.close()
    client_socket.close()
    print("[Python] Ready for next call\n")


def start_agi_server():
    server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    server.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    server.bind(('0.0.0.0', 4573))
    server.listen(5)
    print("[Python] FastAGI server running on port 4573...")

    while True:
        client, addr = server.accept()
        client_handler = threading.Thread(target=handle_asterisk_call, args=(client,))
        client_handler.start()


if __name__ == "__main__":
    start_agi_server()
